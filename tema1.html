1. Memoria y Structs Anidados

Ponderación: 15% (a: 7.5% | b: 7.5%)

Un nodo validador de red financiera procesa cotizaciones de mercado y transacciones criptográficas en memoria rápida. Dadas las siguientes declaraciones de tipos de datos en lenguaje C:

// Definición Struct Base 
struct stCotizacion {
    float PrecioUSD;           // 4 bytes
    float Variacion24h;        // 4 bytes
    unsigned shortint Volumen; // 2 bytes
    char IndicadorTendencia;   // 1 byte
    char ParTrading[7];        // 7 bytes
};
// Definición Struct Compuesto 
struct stOperacionCripto {
    char HashTransaccion[36];         // 36 bytes
    char BilleteraDestino[42];        // 42 bytes
    stCotizacion DatosMercado;        // anidado
    unsigned shortint Confirmaciones; // 2 bytes
    float ComisionFee;                // 4 bytes
    char TipoOrden;                   // 1 byte
    char RedBlockchain[9];            // 9 bytes
};
Consignas a responder justificando el cálculo en bytes:
  1. a. Indique la cantidad total de bytes ocupados en memoria por la variable simple: stCotizacion miTicker;
  2. b. Indique la cantidad total de memoria en bytes requerida para un vector de 40 transacciones: stOperacionCripto Mempool[40];
💡 Ver verificación de cálculo analítico

• stCotizacion: 4 (float) + 4 (float) + 2 (unsigned short) + 1 (char) + 7 (char[7]) = 18 bytes.

• stOperacionCripto: 36 + 42 + 18 (stCotizacion) + 2 + 4 + 1 + 9 = 112 bytes por registro.

• Mempool[40]: 112 bytes × 40 registros = 4480 bytes.

int misterio1(const int arr[], int n, int k) {
    if (n == 0) {
        return 0; // ¿Qué representa esta condición?

5. Problema Integrador en C++ con TDAs Dinámicos

Ponderación: 45%

Desarrolle un programa completo y modular en C++ estándar que resuelva la codificación y empaquetamiento de cadenas para transmisiones financieras seguras, implementando Tipos de Datos Abstractos dinámicos basados en nodos y punteros.

Reglas de Negocio y Criterios del Problema:

  • 1. Validación de Entrada: La palabra ingresada por el usuario debe tener una longitud comprendida estrictamente entre 5 y 25 caracteres, contener exclusivamente letras alfabéticas sin espacios, números ni caracteres especiales.
  • 2. Partición en Estructuras Dinámicas:
    • Las VOCALES se insertan en una PILA DINÁMICA (LIFO).
    • Las CONSONANTES se insertan en una COLA DINÁMICA (FIFO).
  • 3. Algoritmo de Codificación y Alternancia: La palabra codificada se construye alternando las letras de ambas estructuras tomadas por mitades, iniciando por la PILA:
    1. 1ª Mitad de la Pila (desapilando (N_pila / 2) + (N_pila % 2) elementos).
    2. 1ª Mitad de la Cola (desencolando (N_cola / 2) + (N_cola % 2) elementos).
    3. 2ª Mitad restante de la Pila (hasta vaciarla).
    4. 2ª Mitad restante de la Cola (hasta vaciarla).
  • 4. Vaciado Total: Ambas estructuras deben quedar rigurosamente vacías al finalizar el proceso de codificación.

Ejemplo de Ejecución Demostrado:

Palabra ingresada: criptografia (longitud: 12 letras)

PILA (Vocales: 5 letras)
tope -> [ a ]
        [ i ]
        [ a ]
        [ o ]
fondo-> [ i ]
COLA (Consonantes: 7 letras)
frente -> [ c , r , p , t , g , r , f ] <- fin
Secuencia de Extracción de Mitades:
1) 3 letras desapiladas de Pila: 'a', 'i', 'a'
2) 4 letras desencoladas de Cola: 'c', 'r', 'p', 't'
3) 2 letras restantes de Pila: 'o', 'i'
4) 3 letras restantes de Cola: 'g', 'r', 'f'
→ Palabra Codificada: aiacrptoigrf

#include <iostream>
#include <cstring>
#include <cctype>

using namespace std;

// ==========================================
// TDA PILA DINÁMICA (VOCALES)
// ==========================================
struct NodoPila {
    char dato;
    NodoPila* sig;
};

void Apilar(NodoPila* &tope, char valor) {
    NodoPila* nuevo = new NodoPila();
    nuevo->dato = valor;
    nuevo->sig = tope;
    tope = nuevo;
}

char Desapilar(NodoPila* &tope) {
    if (tope == NULL) return '\0';
    NodoPila* aux = tope;
    char valor = aux->dato;
    tope = tope->sig;
    delete aux;
    return valor;
}

bool PilaVacia(NodoPila* tope) {
    return (tope == NULL);
}

// ==========================================
// TDA COLA DINÁMICA (CONSONANTES)
// ==========================================
struct NodoCola {
    char dato;
    NodoCola* sig;
};

struct Cola {
    NodoCola* frente;
    NodoCola* fin;
};

void InicializarCola(Cola &c) {
    c.frente = NULL;
    c.fin = NULL;
}

void Encolar(Cola &c, char valor) {
    NodoCola* nuevo = new NodoCola();
    nuevo->dato = valor;
    nuevo->sig = NULL;
    if (c.frente == NULL) {
        c.frente = nuevo;
    } else {
        c.fin->sig = nuevo;
    }
    c.fin = nuevo;
}

char Desencolar(Cola &c) {
    if (c.frente == NULL) return '\0';
    NodoCola* aux = c.frente;
    char valor = aux->dato;
    c.frente = c.frente->sig;
    if (c.frente == NULL) c.fin = NULL;
    delete aux;
    return valor;
}

bool ColaVacia(Cola c) {
    return (c.frente == NULL);
}

// ==========================================
// PROTOTIPOS MODULARES
// ==========================================
bool EsVocal(char c);
bool ValidarPalabra(const char palabra[]);
void MostrarMenu(int &opc);
void CargarEstructuras(const char palabra[], NodoPila* &pila, Cola &cola, int &cantVocales, int &cantConsonantes);
void MostrarEstructuras(NodoPila* pila, Cola cola);
void CodificarPalabra(NodoPila* &pila, Cola &cola, int cantVocales, int cantConsonantes, char palabraCodificada[]);

int main() {
    int opcion = -1;
    char palabra[50] = "";
    char palabraCodificada[50] = "";
    NodoPila* pilaVocales = NULL;
    Cola colaConsonantes;
    InicializarCola(colaConsonantes);
    int cantVocales = 0, cantConsonantes = 0;

    do {
        MostrarMenu(opcion);
        switch (opcion) {
            case 1:
                cout << "Ingrese palabra (5 a 25 caracteres): ";
                cin >> palabra;
                if (!ValidarPalabra(palabra)) {
                    cout << "[Error] Palabra no valida." << endl;
                    palabra[0] = '\0';
                } else {
                    cout << "[OK] Palabra aceptada." << endl;
                }
                break;
            case 2:
                if (strlen(palabra) == 0) {
                    cout << "[Aviso] Primero ingrese una palabra valida." << endl;
                } else {
                    CargarEstructuras(palabra, pilaVocales, colaConsonantes, cantVocales, cantConsonantes);
                    cout << "[OK] Letras cargadas en Pila y Cola." << endl;
                }
                break;
            case 3:
                MostrarEstructuras(pilaVocales, colaConsonantes);
                break;
            case 4:
                CodificarPalabra(pilaVocales, colaConsonantes, cantVocales, cantConsonantes, palabraCodificada);
                cout << "Palabra Codificada: " << palabraCodificada << endl;
                break;
            case 0:
                cout << "Fin del programa." << endl;
                break;
            default:
                cout << "Opcion invalida." << endl;
        }
    } while (opcion != 0);

    return 0;
}
